import java.util.Scanner;

public class LibraryManagement {

```
public static void main(String[] args) {

    Scanner sc = new Scanner(System.in);
    int books = 50;
    int choice;

    do {
        System.out.println("\n===== LIBRARY MENU =====");
        System.out.println("1. Issue Book");
        System.out.println("2. Return Book");
        System.out.println("3. Check Availability");
        System.out.println("4. Exit");

        System.out.print("Enter your choice: ");
        choice = sc.nextInt();

        switch (choice) {

            case 1:
                System.out.print("Enter number of books to issue: ");
                int issue = sc.nextInt();

                if (issue > 0 && issue <= books) {
                    books = books - issue;
                    System.out.println("Books Issued Successfully");
                } else {
                    System.out.println("Invalid number or not enough books");
                }
                break;

            case 2:
                System.out.print("Enter number of books to return: ");
                int ret = sc.nextInt();

                if (ret > 0) {
                    books = books + ret;
                    System.out.println("Books Returned Successfully");
                } else {
                    System.out.println("Invalid number");
                }
                break;

            case 3:
                System.out.println("Available Books: " + books);
                break;

            case 4:
                System.out.println("Thank you for using Library System");
                break;

            default:
                System.out.println("Invalid Choice");
        }

    } while (choice != 4);

    sc.close();
}
```

}

class Employee {
String name;
int empId;
String dept;

```
Employee(String name, int empId, String dept) {
    this.name = name;
    this.empId = empId;
    this.dept = dept;
}

void displayDetails() {
    System.out.println("Name: " + name);
    System.out.println("Employee ID: " + empId);
    System.out.println("Department: " + dept);
}
```

}

class PermanentEmployee extends Employee {
double salary;
double allowance;

```
PermanentEmployee(String name, int empId, String dept, double salary, double allowance) {
    super(name, empId, dept);
    this.salary = salary;
    this.allowance = allowance;
}

@Override
void displayDetails() {
    super.displayDetails();
    System.out.println("Salary: " + salary);
    System.out.println("Allowance: " + allowance);
}
```

}

class ContractEmployee extends Employee {
int contractDuration;

```
ContractEmployee(String name, int empId, String dept, int contractDuration) {
    super(name, empId, dept);
    this.contractDuration = contractDuration;
}

@Override
void displayDetails() {
    super.displayDetails();
    System.out.println("Contract Duration: " + contractDuration + " months");
}
```

}

public class EmployeeDemo {
public static void main(String[] args) {

```
    PermanentEmployee p = new PermanentEmployee("Amit", 105, "IT", 65000, 5000);
    ContractEmployee c = new ContractEmployee("Neha", 205, "IT", 12);

    System.out.println("===== PERMANENT EMPLOYEE =====");
    p.displayDetails();

    System.out.println("\n===== CONTRACT EMPLOYEE =====");
    c.displayDetails();
}
```

interface Transaction {
void transactionType();
}

class Account {
int accNo;
String holder;

```
Account(int accNo, String holder) {
    this.accNo = accNo;
    this.holder = holder;
}

void displayDetails() {
    System.out.println("Account No: " + accNo);
    System.out.println("Account Holder: " + holder);
}
```

}

class SavingsAccount extends Account implements Transaction {
double interestRate;

```
SavingsAccount(int accNo, String holder, double interestRate) {
    super(accNo, holder);
    this.interestRate = interestRate;
}

@Override
public void transactionType() {
    System.out.println("Transaction Type: Online Transfer");
}

@Override
void displayDetails() {
    super.displayDetails();
    System.out.println("Interest Rate: " + interestRate + "%");
}
```

}

class CurrentAccount extends Account implements Transaction {
double overdraftLimit;

```
CurrentAccount(int accNo, String holder, double overdraftLimit) {
    super(accNo, holder);
    this.overdraftLimit = overdraftLimit;
}

@Override
public void transactionType() {
    System.out.println("Transaction Type: Cheque");
}

@Override
void displayDetails() {
    super.displayDetails();
    System.out.println("Overdraft Limit: " + overdraftLimit);
}
```

}

public class BankSystem {
public static void main(String[] args) {

```
    SavingsAccount s = new SavingsAccount(1001, "Rohan", 6.5);
    CurrentAccount c = new CurrentAccount(2001, "Meera", 50000);

    System.out.println("===== SAVINGS ACCOUNT =====");
    s.displayDetails();
    s.transactionType();

    System.out.println("\n===== CURRENT ACCOUNT =====");
    c.displayDetails();
    c.transactionType();
}
```

}
